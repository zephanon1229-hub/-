#include <stdio.h>
#include <fcntl.h>
#include <unistd.h>
#include <string.h>

int main(void)
{
    const char msg[] = "Hello, Linux System Call!\n";
    char buf[100] = {0};

    // 1. 파일 생성 및 열기
    int fd = open("hello.txt", O_CREAT | O_RDWR | O_TRUNC, 0644);
    if (fd == -1) { perror("open"); return 1; }

    // 2. 파일에 문자열 쓰기
    if (write(fd, msg, strlen(msg)) == -1) {
        perror("write");
        close(fd);
        return 1;
    }

    // 3. 파일 위치를 처음으로 이동
    if (lseek(fd, 0, SEEK_SET) == -1) {
        perror("lseek");
        close(fd);
        return 1;
    }

    // 4. 파일에서 문자열 읽기
    ssize_t n = read(fd, buf, sizeof(buf) - 1);
    if (n == -1) {
        perror("read");
        close(fd);
        return 1;
    }

    // 5. 읽은 내용 화면에 출력
    printf("File content: %s", buf);

    // 6. 파일 닫기
    if (close(fd) == -1) {
        perror("close");
        return 1;
    }

    return 0;
}
